1. VISÃO GERAL E FILOSOFIA
Definição

TurnoCheck é um checklist operacional web que funciona como memória externa confiável durante o expediente de trabalho noturno (16h–00h).

O sistema permanece aberto em uma aba do navegador durante todo o turno e garante que nenhuma tarefa crítica — especialmente as de fechamento — seja esquecida.

Princípio Mais Importante

“O sistema deve parecer mais simples do que a complexidade que resolve.”

Quando houver dúvida entre algo poderoso e algo simples:
sempre escolher o simples.

Problema Central do MVP

Ajudar o usuário a não esquecer tarefas importantes durante o turno.

Toda funcionalidade que não contribuir diretamente para isso está fora do MVP.

Perfil do Usuário

O sistema deve funcionar perfeitamente para alguém:

cansado
distraído
com pressa
encerrando o expediente
usando o sistema rapidamente entre tarefas
2. STACK OFICIAL

Esta stack é definitiva para o MVP.

Camada	Tecnologia
Framework	Next.js App Router
Linguagem	TypeScript
UI	React
Estilo	TailwindCSS
Estado	Zustand
Backend	Supabase
Banco	PostgreSQL
Deploy	Vercel
Autenticação	❌ Não implementar no MVP
Decisão Crítica — Supabase é a Fonte Oficial

O localStorage pode existir apenas como cache temporário.

A fonte oficial dos dados é o Supabase desde o início.

Estratégia Offline (MVP)

O sistema NÃO é offline-first no MVP.

Porém:

atualizações visuais devem ocorrer instantaneamente (optimistic UI)
falhas de sincronização devem exibir toast de erro
alterações temporárias podem permanecer no Zustand até retry manual
não implementar sincronização offline complexa no MVP

8. SCHEMA SUPABASE
create extension if not exists citext;

-- FUNÇÃO UPDATED_AT

create or replace function update_updated_at_column()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

-- WORKSPACES

create table workspaces (
  id uuid primary key default gen_random_uuid(),

  name text not null,

  slug citext not null unique,

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- SECTIONS

create table sections (
  id uuid primary key default gen_random_uuid(),

  workspace_id uuid not null
    references workspaces(id)
    on delete cascade,

  name text not null,

  "order" integer not null default 0,

  color text,

  is_closing_section boolean not null default false,

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- TURNOS

create table turnos (
  id uuid primary key default gen_random_uuid(),

  workspace_id uuid not null
    references workspaces(id)
    on delete cascade,

  started_at timestamptz not null default now(),

  finished_at timestamptz,

  status text not null default 'active'
    check (status in ('active', 'completed', 'archived')),

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- APENAS 1 TURNO ATIVO

create unique index uniq_active_turno_per_workspace
on turnos(workspace_id)
where status = 'active';

-- RECURRING TASKS

create table recurring_tasks (
  id uuid primary key default gen_random_uuid(),

  workspace_id uuid not null
    references workspaces(id)
    on delete cascade,

  section_id uuid
    references sections(id)
    on delete set null,

  title text not null,

  priority text not null default 'normal'
    check (priority in ('low', 'normal', 'high', 'critical')),

  recurrence_type text not null default 'daily'
    check (recurrence_type in ('daily', 'weekly', 'custom')),

  is_active boolean not null default true,

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- TASKS

create table tasks (
  id uuid primary key default gen_random_uuid(),

  turno_id uuid not null
    references turnos(id)
    on delete cascade,

  section_id uuid
    references sections(id)
    on delete set null,

  recurring_task_id uuid
    references recurring_tasks(id)
    on delete set null,

  title text not null,

  description text,

  priority text not null default 'normal'
    check (priority in ('low', 'normal', 'high', 'critical')),

  status text not null default 'pending'
    check (status in ('pending', 'completed')),

  is_pinned boolean not null default false,

  completed_at timestamptz,

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- ÍNDICES

create index idx_tasks_turno_id
on tasks(turno_id);

create index idx_tasks_section_id
on tasks(section_id);

create index idx_tasks_status
on tasks(status, turno_id);

create index idx_turnos_workspace
on turnos(workspace_id, status);

create index idx_recurring_workspace
on recurring_tasks(workspace_id, is_active);

create index idx_sections_workspace
on sections(workspace_id, "order");

-- TRIGGERS UPDATED_AT

create trigger update_workspaces_updated_at
before update on workspaces
for each row
execute function update_updated_at_column();

create trigger update_sections_updated_at
before update on sections
for each row
execute function update_updated_at_column();

create trigger update_turnos_updated_at
before update on turnos
for each row
execute function update_updated_at_column();

create trigger update_recurring_tasks_updated_at
before update on recurring_tasks
for each row
execute function update_updated_at_column();

create trigger update_tasks_updated_at
before update on tasks
for each row
execute function update_updated_at_column();
